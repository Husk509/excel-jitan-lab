中身を確認する(コードの公開)

アドインの中身(VBAコード)を、すべて公開しています。ファイルを開かなくても、ここで読めます。コードだけをダウンロードして、確認することもできます。

先に、結論です。
  • インターネットへ通信するコードは、含まれていません。
  • 他のプログラムを起動するコード(Shellなど)は、アドイン本体には、含まれていません。
  • 使っている外部の部品は、2つだけです。Scripting.Dictionary(同じ値の照合に使う、辞書)と、ADODB.Stream(CSVの文字コードを読み書きする部品)です。どちらも、Windowsに入っています。

確認のしかた

  1. 画面で読む: 下のファイルを開いて、読みます。ブラウザの検索(Ctrl + F)で、Shell、XMLHTTP、WinHttp、URLDownload、Declare を探してみてください。ネットへの通信や、他のプログラムの起動に使う言葉です。
  2. コードだけをダウンロードする: 下の表の「.bas」は、Excelに取り込むための形式です。「.txt」は、メモ帳などで読むための形式です。どちらも、アドイン本体(.xlam)より、警告が出にくい形です。
  3. 自分のExcelに取り込んで、確認する: 新しい、空のブックを開いて、Alt + F11 でVBAの画面を開きます。「ファイル」→「ファイルのインポート」で、.bas を選びます。ブックは、「マクロ有効ブック(.xlsm)」で保存します。

コードを読んで分からない場合は、無理に使わないでください。会社や学校のパソコンでは、管理者のルールに従ってください。

' Excel 標準の RemoveDuplicates は、この呼び出し方だと「引数が不正」になることがあるので、自分で重複を探して、下の行から削除する
Sub Ribbon_RemoveDuplicates(control As IRibbonControl)
    Dim rng As Range, ans As VbMsgBoxResult, data As Variant, dict As Object
    Dim flags() As Boolean, r As Long, c As Long, key As String, startRow As Long, cnt As Long
    Set rng = NT_SelRange()
    If rng Is Nothing Then MsgBox "セル範囲を選んでから、押してください。", vbExclamation, TITLE: Exit Sub
    If rng.Areas.Count > 1 Then MsgBox "範囲は、1か所だけ選んでください。", vbExclamation, TITLE: Exit Sub
    If rng.Cells.CountLarge = 1 Then Set rng = rng.CurrentRegion
    If rng.Rows.Count < 2 Then MsgBox "2行以上のデータを選んでください。", vbExclamation, TITLE: Exit Sub
    ans = MsgBox("範囲 " & rng.Address(False, False) & " で、すべての列が同じ行を、重複として削除します(最初の1行は残します)。" & vbCrLf & vbCrLf & _
                 "1行目は、見出しですか?" & vbCrLf & _
                 "  はい … 見出し(削除の対象にしない)" & vbCrLf & _
                 "  いいえ … 見出しなし" & vbCrLf & _
                 "  キャンセル … 中止" & vbCrLf & vbCrLf & _
                 "※ 元に戻せません。心配なら、先に、ファイルのコピーを取ってください。", _
                 vbYesNoCancel + vbQuestion + vbDefaultButton3, TITLE)
    If ans = vbCancel Then Exit Sub
    startRow = IIf(ans = vbYes, 2, 1)
    On Error GoTo EH
    data = rng.Value2
    Set dict = CreateObject("Scripting.Dictionary")
    dict.CompareMode = vbTextCompare
    ReDim flags(1 To UBound(data, 1))
    For r = startRow To UBound(data, 1)
        key = ""
        For c = 1 To UBound(data, 2)
            If IsError(data(r, c)) Then
                key = key & "#E" & Chr(1)
            Else
                key = key & CStr(data(r, c)) & Chr(1)
            End If
        Next c
        If dict.Exists(key) Then
            flags(r) = True
            cnt = cnt + 1
        Else
            dict.Add key, 1
        End If
    Next r
    If cnt = 0 Then MsgBox "重複した行は、ありませんでした。", vbInformation, TITLE: Exit Sub
    Application.ScreenUpdating = False
    For r = UBound(data, 1) To startRow Step -1
        If flags(r) Then rng.Rows(r).Delete Shift:=xlUp
    Next r
    Application.ScreenUpdating = True
    MsgBox cnt & " 行の重複を、削除しました。", vbInformation, TITLE
    Exit Sub
EH:
    Application.ScreenUpdating = True
    MsgBox "削除できませんでした。(" & Err.Description & ")", vbExclamation, TITLE
End Sub

' 共通: CSVの読み込み

Private Function Csv_IsUtf8(b() As Byte, ByRef hasHigh As Boolean) As Boolean
    Dim i As Long, n As Long, c As Long, k As Long, j As Long
    n = UBound(b)
    i = 0
    Do While i <= n
        c = b(i)
        If c < &H80 Then
            i = i + 1
        Else
            hasHigh = True
            If c >= &HC2 And c <= &HDF Then
                k = 1
            ElseIf c >= &HE0 And c <= &HEF Then
                k = 2
            ElseIf c >= &HF0 And c <= &HF4 Then
                k = 3
            Else
                Exit Function
            End If
            If i + k > n Then Exit Function
            For j = 1 To k
                If (b(i + j) And &HC0) <> &H80 Then Exit Function
            Next j
            i = i + k + 1
        End If
    Loop
    Csv_IsUtf8 = True
End Function

' 文字コードを自動で見分けて(UTF-8 / Shift-JIS / UTF-16)、全体を文字列で返す
Private Function Csv_ReadText(ByVal path As String, ByRef encName As String) As String
    Dim st As Object, b() As Byte, cs As String, hasHigh As Boolean, s As String
    Set st = CreateObject("ADODB.Stream")
    st.Type = 1
    st.Open
    st.LoadFromFile path
    If st.Size = 0 Then
        st.Close
        encName = "空のファイル"
        Exit Function
    End If
    b = st.Read
    st.Close
    If UBound(b) >= 2 Then
        If b(0) = &HEF And b(1) = &HBB And b(2) = &HBF Then
            cs = "utf-8"
            encName = "UTF-8(BOM付き)"
        End If
    End If
    If Len(cs) = 0 And UBound(b) >= 1 Then
        If b(0) = &HFF And b(1) = &HFE Then
            cs = "unicode"
            encName = "UTF-16"
        End If
    End If
    If Len(cs) = 0 Then
        If Csv_IsUtf8(b, hasHigh) Then
            cs = "utf-8"
            encName = IIf(hasHigh, "UTF-8", "ASCII")
        Else
            cs = "shift_jis"
            encName = "Shift-JIS"
        End If
    End If
    Set st = CreateObject("ADODB.Stream")
    st.Type = 2
    st.Charset = cs
    st.Open
    st.LoadFromFile path
    s = st.ReadText(-1)
    st.Close
    If Len(s) > 0 Then
        If AscW(Left$(s, 1)) = &HFEFF Then s = Mid$(s, 2)
    End If
    Csv_ReadText = s
End Function

Private Function Csv_DetectDelim(ByVal txt As String) As String
    Dim p As Long, head As String, nc As Long, nt As Long, ns As Long
    p = InStr(txt, vbLf)
    If p = 0 Then head = txt Else head = Left$(txt, p - 1)
    nc = Len(head) - Len(Replace(head, ",", ""))
    nt = Len(head) - Len(Replace(head, vbTab, ""))
    ns = Len(head) - Len(Replace(head, ";", ""))
    Csv_DetectDelim = ","
    If nt > nc And nt >= ns Then Csv_DetectDelim = vbTab
    If ns > nc And ns > nt Then Csv_DetectDelim = ";"
End Function

' 1レコード(引用符つきの、改行入りの項目も含む)を、項目に分ける
Private Function Csv_SplitRec(ByVal rec As String, ByVal delim As String) As String()
    Dim res() As String, n As Long, i As Long, L As Long
    Dim ch As String, inQ As Boolean, cur As String
    If InStr(rec, """") = 0 Then
        Csv_SplitRec = Split(rec, delim)
        Exit Function
    End If
    ReDim res(0 To 31)
    n = -1
    L = Len(rec)
    i = 1
    Do While i <= L
        ch = Mid$(rec, i, 1)
        If inQ Then
            If ch = """" Then
                If i < L And Mid$(rec, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQ = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQ = True
        ElseIf ch = delim Then
            n = n + 1
            If n > UBound(res) Then ReDim Preserve res(0 To UBound(res) * 2 + 1)
            res(n) = cur
            cur = ""
        Else
            cur = cur & ch
        End If
        i = i + 1
    Loop
    n = n + 1
    If n > UBound(res) Then ReDim Preserve res(0 To UBound(res) * 2 + 1)
    res(n) = cur
    ReDim Preserve res(0 To n)
    Csv_SplitRec = res
End Function

' 項目1つを、セルに入れる値にする。
' ・=, +, @ で始まる文字は、数式として動かさず、文字のまま入れる(CSVの中の数式が、実行されるのを防ぐ)
' ・001 のように先頭が 0 の番号や、16桁以上の番号は、文字のまま入れる(0や下の桁が消えないように)
Private Function Csv_Cell(ByVal s As String) As Variant
    Dim c1 As String
    If Len(s) = 0 Then Exit Function
    c1 = Left$(s, 1)
    If c1 = "=" Or c1 = "+" Or c1 = "@" Then
        Csv_Cell = "'" & s
    ElseIf c1 = "-" And Not IsNumeric(s) Then
        Csv_Cell = "'" & s
    ElseIf Not (s Like "*[!0-9]*") And ((c1 = "0" And Len(s) > 1) Or Len(s) >= 16) Then
        Csv_Cell = "'" & s
    Else
        Csv_Cell = s
    End If
End Function

' 見出しの名前として使うため、先頭の ' を外した文字にする
Private Function Csv_Plain(ByVal v As Variant) As String
    Dim s As String
    If IsEmpty(v) Then Exit Function
    s = CStr(v)
    If Left$(s, 1) = "'" Then s = Mid$(s, 2)
    Csv_Plain = s
End Function

' CSV全体を、2次元の配列(行, 列)にする。空の行は飛ばす
Private Function Csv_Parse(ByVal txt As String, ByRef nRows As Long, ByRef nCols As Long) As Variant
    Dim delim As String, lines() As String, i As Long, ub As Long, rec As String
    Dim recs() As Variant, cnt As Long, f() As String, r As Long, c As Long
    Dim out() As Variant
    txt = Replace(txt, vbCrLf, vbLf)
    txt = Replace(txt, vbCr, vbLf)
    delim = Csv_DetectDelim(txt)
    lines = Split(txt, vbLf)
    ub = UBound(lines)
    ReDim recs(1 To 1024)
    nCols = 0
    i = 0
    Do While i <= ub
        rec = lines(i)
        i = i + 1
        Do While ((Len(rec) - Len(Replace(rec, """", ""))) Mod 2 = 1) And i <= ub
            rec = rec & vbLf & lines(i)
            i = i + 1
        Loop
        If Len(rec) > 0 Then
            f = Csv_SplitRec(rec, delim)
            cnt = cnt + 1
            If cnt > UBound(recs) Then ReDim Preserve recs(1 To UBound(recs) * 2)
            recs(cnt) = f
            If UBound(f) + 1 > nCols Then nCols = UBound(f) + 1
        End If
    Loop
    nRows = cnt
    If cnt = 0 Then
        Csv_Parse = Empty
        Exit Function
    End If
    ReDim out(1 To cnt, 1 To nCols)
    For r = 1 To cnt
        f = recs(r)
        For c = 0 To UBound(f)
            out(r, c + 1) = Csv_Cell(f(c))
        Next c
        recs(r) = Empty
    Next r
    Csv_Parse = out
End Function

Private Function Csv_UniqueSheetName(ByVal baseName As String) As String
    Dim nm As String, n As Long, ws As Object, hit As Boolean
    nm = baseName
    n = 1
    Do
        hit = False
        For Each ws In ActiveWorkbook.Sheets
            If StrComp(ws.Name, nm, vbTextCompare) = 0 Then hit = True: Exit For
        Next ws
        If Not hit Then Exit Do
        n = n + 1
        nm = baseName & "(" & n & ")"
    Loop
    Csv_UniqueSheetName = nm
End Function

        .Filters.Add "すべてのファイル", "*.*"
        If .Show <> -1 Then Exit Sub
        nF = .SelectedItems.Count
        ReDim paths(1 To nF)
        For k = 1 To nF
            paths(k) = .SelectedItems(k)
        Next k
    End With
    ' ファイル名の順にそろえる(毎回、同じ順番になるように)
    For k = 1 To nF - 1
        For i = k + 1 To nF
            If StrComp(paths(i), paths(k), vbTextCompare) < 0 Then tmp = paths(k): paths(k) = paths(i): paths(i) = tmp
        Next i
    Next k

    ans = MsgBox("選んだ " & nF & " 個のCSVの1行目は、見出し(列の名前)ですか?" & vbCrLf & vbCrLf & _
                 "はい … 見出しの名前で列をそろえて、結合します(列の並びが違っても大丈夫)" & vbCrLf & _
                 "いいえ … 見出しなし。左から順に、積み重ねます" & vbCrLf & _
                 "キャンセル … 中止", vbYesNoCancel + vbQuestion, TITLE)
    If ans = vbCancel Then Exit Sub
    hasHdr = (ans = vbYes)
    addSrc = (MsgBox("先頭の列に、元のファイル名を入れますか?", vbYesNo + vbQuestion, TITLE) = vbYes)
    Set ws = Csv_DoMerge(paths, hasHdr, addSrc, msg)
    MsgBox msg, IIf(ws Is Nothing, vbExclamation, vbInformation), TITLE
End Sub

' 結合の本体。成功したら、新しいシートを返す(失敗したら Nothing)。msg に、ユーザーへの報告を入れる
' paths は、1 から始まる配列
Private Function Csv_DoMerge(paths() As String, ByVal hasHdr As Boolean, ByVal addSrc As Boolean, ByRef msg As String) As Worksheet
    Dim nF As Long, k As Long, sc As Long
    Dim names() As String, encs() As String
    Dim datas() As Variant, maps() As Variant, rowsN() As Long, colsN() As Long
    Dim hdrDict As Object, used As Object, hdrList() As String, nHdr As Long
    Dim m() As Long, c As Long, nm As String, key As String, cnt As Long, txt As String
    Dim totalRows As Long, startRow As Long, outCols As Long, nR As Long, r As Long, nxt As Long
    Dim ws As Worksheet, a() As Variant, src As Variant, dataRows As Long
    Dim diffCols As Boolean, shown As Long
    nF = UBound(paths)
    sc = IIf(addSrc, 1, 0)
    startRow = IIf(hasHdr, 2, 1)

    On Error GoTo EH
    Application.ScreenUpdating = False
    ReDim datas(1 To nF): ReDim maps(1 To nF)
    ReDim rowsN(1 To nF): ReDim colsN(1 To nF)
    ReDim names(1 To nF): ReDim encs(1 To nF)

    ' 読み込み(CSVは書き換えない)
    For k = 1 To nF
        names(k) = Mid$(paths(k), InStrRev(paths(k), "\") + 1)
        txt = Csv_ReadText(paths(k), encs(k))
        datas(k) = Csv_Parse(txt, rowsN(k), colsN(k))
        txt = ""
    Next k

    ' 列の対応づけ(見出しの名前、または左からの順)
    Set hdrDict = CreateObject("Scripting.Dictionary")
    hdrDict.CompareMode = vbTextCompare
    ReDim hdrList(1 To 64)
    For k = 1 To nF
        If colsN(k) > 0 Then
            src = datas(k)
            ReDim m(1 To colsN(k))
            If hasHdr Then
                Set used = CreateObject("Scripting.Dictionary")
                used.CompareMode = vbTextCompare
                For c = 1 To colsN(k)
                    nm = Trim$(Csv_Plain(src(1, c)))
                    If Len(nm) = 0 Then nm = "(列" & c & ")"
                    key = nm
                    cnt = 1
                    Do While used.Exists(key)
                        cnt = cnt + 1
                        key = nm & "(" & cnt & ")"
                    Loop
                    used.Add key, 1
                    If Not hdrDict.Exists(key) Then
                        nHdr = nHdr + 1
                        hdrDict.Add key, nHdr
                        If nHdr > UBound(hdrList) Then ReDim Preserve hdrList(1 To nHdr * 2)
                        hdrList(nHdr) = key
                    End If
                    m(c) = hdrDict(key)
                Next c
            Else
                For c = 1 To colsN(k)
                    m(c) = c
                Next c
                If colsN(k) > nHdr Then nHdr = colsN(k)
            End If
            maps(k) = m
        End If
    Next k
    If nHdr = 0 Then
        Application.ScreenUpdating = True
        msg = "読み込める行がありませんでした。"
        Exit Function
    End If

    ' 行数の確認
    totalRows = IIf(hasHdr, 1, 0)
    For k = 1 To nF
        If colsN(k) > 0 And rowsN(k) >= startRow Then totalRows = totalRows + rowsN(k) - startRow + 1
        If colsN(k) <> nHdr Then diffCols = True
    Next k
    outCols = nHdr + sc
    If totalRows > MAX_ROWS Or outCols > 16384 Then
        Application.ScreenUpdating = True
        msg = "Excelの上限(" & Format(MAX_ROWS, "#,##0") & " 行 / 16384 列)を超えるため、結合できません。"
        Exit Function
    End If

    ' 書き出し(新しいシートに)
    If Workbooks.Count = 0 Then Workbooks.Add
    Set ws = ActiveWorkbook.Worksheets.Add(After:=ActiveWorkbook.Sheets(ActiveWorkbook.Sheets.Count))
    ws.Name = Csv_UniqueSheetName("CSV結合")
    nxt = 1
    If hasHdr Then
        ReDim a(1 To 1, 1 To outCols)
        If addSrc Then a(1, 1) = "元ファイル"
        For c = 1 To nHdr
            a(1, sc + c) = Csv_Cell(hdrList(c))
        Next c
        ws.Cells(1, 1).Resize(1, outCols).Value2 = a
        nxt = 2
    End If
    For k = 1 To nF
        If colsN(k) > 0 And rowsN(k) >= startRow Then
            src = datas(k)
            m = maps(k)
            nR = rowsN(k) - startRow + 1
            ReDim a(1 To nR, 1 To outCols)
            For r = 1 To nR
                If addSrc Then a(r, 1) = Csv_Cell(names(k))
                For c = 1 To colsN(k)
                    a(r, sc + m(c)) = src(r + startRow - 1, c)
                Next c
            Next r
            ws.Cells(nxt, 1).Resize(nR, outCols).Value2 = a
            nxt = nxt + nR
            dataRows = dataRows + nR
        End If
        datas(k) = Empty
    Next k
    If hasHdr Then ws.Rows(1).Font.Bold = True
    If totalRows <= 50000 Then ws.Columns.AutoFit
    ws.Activate
    ws.Cells(1, 1).Select
    Application.ScreenUpdating = True

    ' 報告(検算つき)
    msg = "結合しました。新しいシート「" & ws.Name & "」を見てください。" & vbCrLf & vbCrLf
    For k = 1 To nF
        If shown >= 15 Then Exit For
        shown = shown + 1
        msg = msg & names(k) & " … " & encs(k) & " / " & IIf(colsN(k) > 0, rowsN(k) - IIf(hasHdr, 1, 0), 0) & " 行" & vbCrLf
    Next k
    If nF > shown Then msg = msg & "ほか " & (nF - shown) & " 個" & vbCrLf
    msg = msg & vbCrLf & "合計 " & Format(dataRows, "#,##0") & " 行" & IIf(hasHdr, "(見出しを除く)", "") & " / " & nHdr & " 列" & vbCrLf
    If (nxt - 1) = totalRows Then
        msg = msg & "検算: 書き込んだ行数が、各ファイルの行数の合計と一致しました。"
    Else
        msg = msg & "検算: 行数が一致しません。結果を確認してください。"
    End If
    If diffCols Then msg = msg & vbCrLf & "※ ファイルによって列の数が違います。ないところは空白です。"
    Set Csv_DoMerge = ws
    Exit Function
EH:
    Application.ScreenUpdating = True
    msg = "うまくいきませんでした。" & vbCrLf & Err.Description
    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Delete
        Application.DisplayAlerts = True
    End If
End Function

' 列の並べ替えの本体。成功したら、新しいシートを返す(失敗したら Nothing)。msg に、ユーザーへの報告を入れる
Private Function Csv_DoColumns(ByVal ws As Object, ByVal spec As String, ByRef msg As String) As Worksheet
    Dim ur As Range, nc As Long, i As Long, j As Long
    Dim toks() As String, idx() As Long, t As String, tn As String, n As Long
    Dim found As Long, bad As String, nw As Worksheet
    Set ur = ws.UsedRange
    nc = ur.Columns.Count
    spec = Replace(Replace(spec, "、", ","), "，", ",")
    toks = Split(spec, ",")
    ReDim idx(1 To UBound(toks) + 1)
    n = 0
    For i = 0 To UBound(toks)
        t = Trim$(toks(i))
        If Len(t) > 0 Then
            tn = StrConv(t, vbNarrow)
            found = 0
            If IsNumeric(tn) Then
                If CDbl(tn) >= 1 And CDbl(tn) <= nc And CDbl(tn) = Int(CDbl(tn)) Then found = CLng(tn)
            Else
                For j = 1 To nc
                    If StrComp(Trim$(CStr(ur.Cells(1, j).Value)), t, vbTextCompare) = 0 Then found = j: Exit For
                Next j
                If found = 0 And Len(tn) <= 3 And Not (tn Like "*[!A-Za-z]*") Then
                    j = ws.Range(tn & "1").Column - ur.Column + 1
                    If j >= 1 And j <= nc Then found = j
                End If
            End If
            If found = 0 Then
                bad = bad & "「" & t & "」"
            Else
                n = n + 1
                idx(n) = found
            End If
        End If
    Next i
    If Len(bad) > 0 Then
        msg = "次の指定は、列として見つかりませんでした: " & bad & vbCrLf & "もう一度、やり直してください。"
        Exit Function
    End If
    If n = 0 Then
        msg = "列が指定されていません。"
        Exit Function
    End If

    On Error GoTo EH
    Application.ScreenUpdating = False
    Set nw = ActiveWorkbook.Worksheets.Add(After:=ActiveWorkbook.Sheets(ActiveWorkbook.Sheets.Count))
    nw.Name = Csv_UniqueSheetName("列並べ替え")
    For j = 1 To n
        ur.Columns(idx(j)).Copy
        nw.Cells(1, j).PasteSpecial Paste:=xlPasteValuesAndNumberFormats
        nw.Columns(j).ColumnWidth = ur.Columns(idx(j)).ColumnWidth
    Next j
    Application.CutCopyMode = False
    nw.Activate
    nw.Cells(1, 1).Select
    Application.ScreenUpdating = True
    msg = "新しいシート「" & nw.Name & "」に、" & n & " 列を並べました。元のシートは、そのままです。" & vbCrLf & "(数式は、結果の値になります)"
    Set Csv_DoColumns = nw
    Exit Function
EH:
    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    msg = "うまくいきませんでした。" & vbCrLf & Err.Description
End Function

' 3. 開いているシートを、CSVで保存
' ---------------------------------------------------------------
Private Function Csv_Field(ByVal v As Variant, ByRef errs As Long) As String
    Dim s As String
    Select Case VarType(v)
        Case vbEmpty, vbNull
            s = ""
        Case vbError
            s = ""
            errs = errs + 1
        Case vbDate
            If CDbl(v) < 1 Then
                s = Format$(v, "hh:nn:ss")
            ElseIf CDbl(v) = Int(CDbl(v)) Then
                s = Format$(v, "yyyy/mm/dd")
            Else
                s = Format$(v, "yyyy/mm/dd hh:nn:ss")
            End If
        Case vbBoolean
            s = IIf(v, "TRUE", "FALSE")
        Case Else
            s = CStr(v)
    End Select
    If InStr(s, ",") > 0 Or InStr(s, """") > 0 Or InStr(s, vbLf) > 0 Or InStr(s, vbCr) > 0 _
       Or Left$(s, 1) = " " Or Right$(s, 1) = " " Then
        s = """" & Replace(s, """", """""") & """"
    End If
    Csv_Field = s
End Function

' 保存の本体。cs は "shift_jis" か "utf-8"。成功したら True。msg に、ユーザーへの報告を入れる
Private Function Csv_DoSave(ByVal ws As Object, ByVal path As String, ByVal cs As String, ByRef msg As String) As Boolean
    Dim ur As Range, data As Variant, one(1 To 1, 1 To 1) As Variant
    Dim r As Long, c As Long, nR As Long, nc As Long, f() As String, lines() As String
    Dim errs As Long, txt As String, back As String, enc As String
    Set ur = ws.UsedRange
    On Error GoTo EH
    Application.ScreenUpdating = False
    If ur.Cells.CountLarge = 1 Then
        one(1, 1) = ur.Value
        data = one
    Else
        data = ur.Value
    End If
    nR = UBound(data, 1)
    nc = UBound(data, 2)
    ReDim f(1 To nc)
    ReDim lines(1 To nR)
    For r = 1 To nR
        For c = 1 To nc
            f(c) = Csv_Field(data(r, c), errs)
        Next c
        lines(r) = Join(f, ",")
    Next r
    txt = Join(lines, vbCrLf) & vbCrLf
    Csv_WriteText path, txt, cs
    Application.ScreenUpdating = True

    msg = "保存しました。" & vbCrLf & path & vbCrLf & nR & " 行 / " & nc & " 列"
    If cs = "shift_jis" Then
        back = Csv_ReadText(path, enc)
        If back <> txt Then msg = msg & vbCrLf & vbCrLf & "※ Shift-JIS にない文字が、「?」に変わった可能性があります。UTF-8 でも保存できます。"
    End If
    If errs > 0 Then msg = msg & vbCrLf & vbCrLf & "※ エラー値のセルが " & errs & " 個ありました(空白で保存しました)。"
    Csv_DoSave = True
    Exit Function
EH:
    Application.ScreenUpdating = True
    msg = "うまくいきませんでした。" & vbCrLf & Err.Description
End Function

Private Const TITLE As String = "突合ツール"

' ---------------------------------------------------------------

' セルの値を、比べるための文字にする(前後の空白と、全角の空白をなくす)
Private Function Rc_Norm(ByVal v As Variant) As String
    Dim s As String
    If IsError(v) Then Rc_Norm = "#ERR": Exit Function
    If IsEmpty(v) Then Exit Function
    ' 日付の値は、46336 のような数字ではなく、日付の文字にする(時刻が入っているときだけ、時刻も付ける)
    If VarType(v) = vbDate Then
        If v = Int(v) Then
            Rc_Norm = Format$(v, "yyyy/mm/dd")
        Else
            Rc_Norm = Format$(v, "yyyy/mm/dd hh:nn")
        End If
        Exit Function
    End If
    s = CStr(v)
    s = Replace(s, ChrW(&H3000), " ")
    Rc_Norm = Trim$(s)
End Function

' 値が同じか。数字として読めるものは、数字で比べる(tol 以内の差なら、同じ)
Private Function Rc_Same(ByVal a As Variant, ByVal b As Variant, ByVal tol As Double) As Boolean
    Dim sa As String, sb As String
    sa = Rc_Norm(a)
    sb = Rc_Norm(b)
    If sa = sb Then Rc_Same = True: Exit Function
    If Len(sa) > 0 And Len(sb) > 0 And Len(sa) <= 15 And Len(sb) <= 15 Then
        If IsNumeric(sa) And IsNumeric(sb) Then
            Rc_Same = (Abs(CDbl(sa) - CDbl(sb)) <= tol + 0.0000000001)
        End If
    End If
End Function

' キーの文字(大文字小文字は区別しない)。キーがすべて空白なら、blank を True にする
Private Function Rc_Key(d As Variant, ByVal r As Long, cols() As Long, ByRef blank As Boolean) As String
    Dim k As Long, s As String, part As String, anyVal As Boolean
    For k = 1 To UBound(cols)
        part = Rc_Norm(d(r, cols(k)))
        If Len(part) > 0 Then anyVal = True
        s = s & LCase$(part) & Chr(1)
    Next k
    blank = Not anyVal
    Rc_Key = s
End Function

Private Function Rc_Show(d As Variant, ByVal r As Long, cols() As Long) As String
    Dim k As Long, s As String
    For k = 1 To UBound(cols)
        If k > 1 Then s = s & " / "
        s = s & Rc_Norm(d(r, cols(k)))
    Next k
    Rc_Show = s
End Function

' 1行ぶんの中身を、「列名=値; 列名=値」の文字にする(長いときは、途中まで)
Private Function Rc_Desc(d As Variant, ByVal r As Long, hdr() As String, ByVal nc As Long) As String
    Dim c As Long, s As String, v As String
    For c = 1 To nc
        v = Rc_Norm(d(r, c))
        If Len(v) > 0 Then
            If Len(s) > 0 Then s = s & "; "
            s = s & hdr(c) & "=" & v
        End If
        If Len(s) > 250 Then s = Left$(s, 250) & "...": Exit For
    Next c
    Rc_Desc = s
End Function

' 結果の文字が、数式として動かないようにする
Private Function Rc_Safe(ByVal s As String) As String
    Dim c1 As String
    If Len(s) = 0 Then Exit Function
    c1 = Left$(s, 1)
    If c1 = "=" Or c1 = "+" Or c1 = "@" Or c1 = "-" Then Rc_Safe = "'" & s Else Rc_Safe = s
End Function

Private Sub Rc_Put(res() As Variant, ByRef cnt As Long, ByVal a As String, ByVal b As String, ByVal c As String, ByVal d As Variant, ByVal e As Variant)
    cnt = cnt + 1
    res(cnt, 1) = a
    res(cnt, 2) = Rc_Safe(b)
    res(cnt, 3) = Rc_Safe(c)
    res(cnt, 4) = d
    res(cnt, 5) = e
End Sub

Private Function Rc_UniqueSheetName(ByVal baseName As String) As String
    Dim nm As String, n As Long, ws As Object, hit As Boolean
    nm = baseName
    n = 1
    Do
        hit = False
        For Each ws In ActiveWorkbook.Sheets
            If StrComp(ws.Name, nm, vbTextCompare) = 0 Then hit = True: Exit For
        Next ws
        If Not hit Then Exit Do
        n = n + 1
        nm = baseName & "(" & n & ")"
    Loop
    Rc_UniqueSheetName = nm
End Function

        hdr = hdr & i & "=" & Left$(Rc_Norm(rA.Cells(1, i).Value2), 10) & "  "
    Next i
    keySpec = InputBox("照らし合わせるキー(伝票番号、顧客コードなど)の列を、見出しの名前か番号で入れてください。" & vbCrLf & _
                       "2つ以上は、カンマで区切ります。見出しの名前は、2つの表で同じである必要があります。" & vbCrLf & vbCrLf & _
                       "1つ目の表の列: " & hdr, TITLE & " (3/4)", Rc_Norm(rA.Cells(1, 1).Value2))
    If Len(Trim$(keySpec)) = 0 Then Exit Sub
    tolS = InputBox("数字の差が、いくつ以内なら「同じ」とみなしますか?" & vbCrLf & _
                    "0 なら、1円の違いも、差として出します。端数の違いを許すなら、1 などを入れます。", TITLE & " (4/4)", "0")
    If Len(Trim$(tolS)) = 0 Then Exit Sub
    tolS = StrConv(tolS, vbNarrow)
    If Not IsNumeric(tolS) Then MsgBox "数字で入れてください。", vbExclamation, TITLE: Exit Sub
    tol = Abs(CDbl(tolS))

    Set ws = Recon_Do(rA, rB, keySpec, tol, msg)
    MsgBox msg, IIf(ws Is Nothing, vbExclamation, vbInformation), TITLE

' ---------------------------------------------------------------
' 本体。成功したら、結果のシートを返す(失敗したら Nothing)。msg に、ユーザーへの報告を入れる
' ---------------------------------------------------------------
Private Function Recon_Do(ByVal rA As Range, ByVal rB As Range, ByVal keySpec As String, ByVal tol As Double, ByRef msg As String) As Worksheet
    Dim dA As Variant, dB As Variant, nA As Long, nB As Long, cA As Long, cB As Long
    Dim hA() As String, hB() As String, dictHA As Object, dictHB As Object
    Dim toks() As String, i As Long, c As Long, r As Long, t As String, tn As String, idx As Long
    Dim nk As Long, kA() As Long, kB() As Long, kName() As String, bad As String
    Dim pA() As Long, pB() As Long, pName() As String, np As Long, isKey As Boolean, p As Long
    Dim onlyA As String, onlyB As String, nm As String
    Dim dictKB As Object, seenA As Object, bState() As Long, usedB() As Boolean
    Dim k As String, blank As Boolean, br As Long, diffs As String, nd As Long
    Dim res() As Variant, cnt As Long, cap As Long
    Dim nSame As Long, nDiff As Long, nOnlyA As Long, nOnlyB As Long, dupA As Long, dupB As Long, emptyA As Long, emptyB As Long
    Dim ws As Worksheet, last As Long, hd(1 To 1, 1 To 5) As Variant, sm(1 To 12, 1 To 2) As Variant
    Dim okA As Boolean, okB As Boolean

    On Error GoTo EH
    If rA.Rows.Count < 2 Or rB.Rows.Count < 2 Then
        msg = "表は、見出しの行のほかに、データが1行以上必要です。"
        Exit Function
    End If
    ' .Value2 だと、日付が数字(シリアル値)になるので、日付の型が残る .Value で読む
    dA = rA.Value
    dB = rB.Value
    nA = UBound(dA, 1): cA = UBound(dA, 2)
    nB = UBound(dB, 1): cB = UBound(dB, 2)

    ' 見出し
    Set dictHA = CreateObject("Scripting.Dictionary"): dictHA.CompareMode = vbTextCompare
    Set dictHB = CreateObject("Scripting.Dictionary"): dictHB.CompareMode = vbTextCompare
    ReDim hA(1 To cA): ReDim hB(1 To cB)
    For c = 1 To cA
        hA(c) = Rc_Norm(dA(1, c))
        If Len(hA(c)) > 0 Then If Not dictHA.Exists(hA(c)) Then dictHA.Add hA(c), c
    Next c
    For c = 1 To cB
        hB(c) = Rc_Norm(dB(1, c))
        If Len(hB(c)) > 0 Then If Not dictHB.Exists(hB(c)) Then dictHB.Add hB(c), c
    Next c

    ' キーの列
    keySpec = Replace(Replace(keySpec, "、", ","), "，", ",")
    toks = Split(keySpec, ",")
    ReDim kA(1 To UBound(toks) + 1): ReDim kB(1 To UBound(toks) + 1): ReDim kName(1 To UBound(toks) + 1)
    For i = 0 To UBound(toks)
        t = Trim$(toks(i))
        If Len(t) > 0 Then
            tn = StrConv(t, vbNarrow)
            If Not dictHA.Exists(t) And IsNumeric(tn) Then
                idx = CLng(Val(tn))
                If idx >= 1 And idx <= cA Then t = hA(idx)
            End If
            If Len(t) > 0 And dictHA.Exists(t) And dictHB.Exists(t) Then
                nk = nk + 1
                kA(nk) = dictHA(t): kB(nk) = dictHB(t): kName(nk) = t
            Else
                bad = bad & "「" & t & "」"
            End If
        End If
    Next i
    If Len(bad) > 0 Or nk = 0 Then
        msg = "キーの列が見つかりませんでした: " & bad & vbCrLf & "2つの表に、同じ名前の見出しがあるか、確認してください。"
        Exit Function
    End If
    ReDim Preserve kA(1 To nk): ReDim Preserve kB(1 To nk): ReDim Preserve kName(1 To nk)

    ' 比べる列(キー以外で、見出しの名前が同じ列)
    ReDim pA(1 To cA): ReDim pB(1 To cA): ReDim pName(1 To cA)
    For c = 1 To cA
        nm = hA(c)
        If Len(nm) > 0 Then
            If dictHA(nm) = c Then
                isKey = False
                For i = 1 To nk
                    If StrComp(kName(i), nm, vbTextCompare) = 0 Then isKey = True
                Next i
                If Not isKey Then
                    If dictHB.Exists(nm) Then
                        np = np + 1
                        pA(np) = c: pB(np) = dictHB(nm): pName(np) = nm
                    Else
                        onlyA = onlyA & "「" & nm & "」"
                    End If
                End If
            End If
        End If
    Next c
    For c = 1 To cB
        nm = hB(c)
        If Len(nm) > 0 Then
            If dictHB(nm) = c And Not dictHA.Exists(nm) Then onlyB = onlyB & "「" & nm & "」"
        End If
    Next c

    ' 表Bのキーを、先に集める
    Set dictKB = CreateObject("Scripting.Dictionary")
    Set seenA = CreateObject("Scripting.Dictionary")
    ReDim bState(2 To nB): ReDim usedB(2 To nB)
    For r = 2 To nB
        k = Rc_Key(dB, r, kB, blank)
        If blank Then
            bState(r) = 2
            emptyB = emptyB + 1
        ElseIf dictKB.Exists(k) Then
            bState(r) = 1
            dupB = dupB + 1
        Else
            dictKB.Add k, r
        End If
    Next r

    cap = (nA - 1) + (nB - 1)
    ReDim res(1 To cap, 1 To 5)

    ' 表Aを順に見る
    For r = 2 To nA
        k = Rc_Key(dA, r, kA, blank)
        If blank Then
            emptyA = emptyA + 1
            Rc_Put res, cnt, "キー空(A)", "(キーが空白)", Rc_Desc(dA, r, hA, cA), rA.Row + r - 1, ""
        ElseIf seenA.Exists(k) Then
            dupA = dupA + 1
            Rc_Put res, cnt, "重複(A)", Rc_Show(dA, r, kA), "同じキーが、" & (rA.Row + seenA(k) - 1) & "行目にもあります", rA.Row + r - 1, ""
        Else
            seenA.Add k, r
            If dictKB.Exists(k) Then
                br = dictKB(k)
                usedB(br) = True
                diffs = ""
                nd = 0
                For p = 1 To np
                    If Not Rc_Same(dA(r, pA(p)), dB(br, pB(p)), tol) Then
                        nd = nd + 1
                        If nd > 1 Then diffs = diffs & "; "
                        diffs = diffs & pName(p) & ": " & Rc_Norm(dA(r, pA(p))) & " → " & Rc_Norm(dB(br, pB(p)))
                    End If
                Next p
                If nd = 0 Then
                    nSame = nSame + 1
                Else
                    nDiff = nDiff + 1
                    Rc_Put res, cnt, "差異", Rc_Show(dA, r, kA), diffs, rA.Row + r - 1, rB.Row + br - 1
                End If
            Else
                nOnlyA = nOnlyA + 1
                Rc_Put res, cnt, "Aのみ", Rc_Show(dA, r, kA), Rc_Desc(dA, r, hA, cA), rA.Row + r - 1, ""
            End If
        End If
    Next r

    ' 表Bの、残り
    For r = 2 To nB
        Select Case bState(r)
            Case 2
                Rc_Put res, cnt, "キー空(B)", "(キーが空白)", Rc_Desc(dB, r, hB, cB), "", rB.Row + r - 1
            Case 1
                k = Rc_Key(dB, r, kB, blank)
                Rc_Put res, cnt, "重複(B)", Rc_Show(dB, r, kB), "同じキーが、" & (rB.Row + dictKB(k) - 1) & "行目にもあります", "", rB.Row + r - 1
            Case Else
                If Not usedB(r) Then
                    nOnlyB = nOnlyB + 1
                    Rc_Put res, cnt, "Bのみ", Rc_Show(dB, r, kB), Rc_Desc(dB, r, hB, cB), "", rB.Row + r - 1
                End If
        End Select
    Next r

    ' 検算
    okA = ((nA - 1) = nSame + nDiff + nOnlyA + dupA + emptyA)
    okB = ((nB - 1) = nSame + nDiff + nOnlyB + dupB + emptyB)

    ' 結果のシート
    Application.ScreenUpdating = False
    Set ws = ActiveWorkbook.Worksheets.Add(After:=ActiveWorkbook.Sheets(ActiveWorkbook.Sheets.Count))
    ws.Name = Rc_UniqueSheetName("突合結果")
    hd(1, 1) = "結果"
    hd(1, 2) = "キー(" & Join(kName, " / ") & ")"
    hd(1, 3) = "内容"
    hd(1, 4) = "Aの行"
    hd(1, 5) = "Bの行"
    ws.Range("A1:E1").Value2 = hd
    last = cnt + 1
    If cnt > 0 Then
        ws.Range("A2").Resize(cap, 5).Value2 = res
    Else
        ws.Range("A2").Value2 = "(すべて一致しました)"
        last = 2
    End If
    sm(1, 1) = "表Aの行数(見出し除く)": sm(1, 2) = nA - 1
    sm(2, 1) = "表Bの行数(見出し除く)": sm(2, 2) = nB - 1
    sm(3, 1) = "一致": sm(3, 2) = nSame
    sm(4, 1) = "差異": sm(4, 2) = nDiff
    sm(5, 1) = "Aのみ": sm(5, 2) = nOnlyA
    sm(6, 1) = "Bのみ": sm(6, 2) = nOnlyB
    sm(7, 1) = "重複(A)": sm(7, 2) = dupA
    sm(8, 1) = "重複(B)": sm(8, 2) = dupB
    sm(9, 1) = "キー空(A)": sm(9, 2) = emptyA
    sm(10, 1) = "キー空(B)": sm(10, 2) = emptyB
    sm(11, 1) = "検算(A)": sm(11, 2) = IIf(okA, "OK", "不一致")
    sm(12, 1) = "検算(B)": sm(12, 2) = IIf(okB, "OK", "不一致")
    ws.Range("G1:H12").Value2 = sm
    ws.Range("A1:E1").Font.Bold = True
    ws.Range("A1:E1").Interior.Color = RGB(217, 217, 217)
    ws.Range("G1:G12").Font.Bold = True
    With ws.Range("A2:E" & last).FormatConditions
        .Delete
        .Add(2, , "=$A2=""差異""").Interior.Color = RGB(252, 228, 214)
        .Add(2, , "=$A2=""Aのみ""").Interior.Color = RGB(255, 242, 204)
        .Add(2, , "=$A2=""Bのみ""").Interior.Color = RGB(221, 235, 247)
        .Add(2, , "=LEFT($A2,2)=""重複""").Interior.Color = RGB(237, 237, 237)
        .Add(2, , "=LEFT($A2,2)=""キー""").Interior.Color = RGB(237, 237, 237)
    End With
    ws.Columns("A:H").AutoFit
    If ws.Columns(3).ColumnWidth > 70 Then ws.Columns(3).ColumnWidth = 70
    On Error Resume Next
    ws.Range("A1:E" & last).AutoFilter
    ws.Activate
    ActiveWindow.SplitColumn = 0
    ActiveWindow.SplitRow = 1
    ActiveWindow.FreezePanes = True
    ws.Range("A1").Select
    On Error GoTo EH
    Application.ScreenUpdating = True

    msg = "突合しました。新しいシート「" & ws.Name & "」を見てください。" & vbCrLf & vbCrLf & _
          "表A " & (nA - 1) & " 行 / 表B " & (nB - 1) & " 行 / キー: " & Join(kName, " + ") & vbCrLf & _
          "一致 " & nSame & " / 差異 " & nDiff & " / Aのみ " & nOnlyA & " / Bのみ " & nOnlyB & vbCrLf & _
          "重複 A " & dupA & ", B " & dupB & " / キー空白 A " & emptyA & ", B " & emptyB & vbCrLf
    msg = msg & "検算: " & IIf(okA And okB, "行数がすべて合いました。", "行数が合いません。結果を確認してください。")
    If np = 0 Then msg = msg & vbCrLf & "※ 比べる列がありません(キーの有無だけ見ました)。"
    If Len(onlyA) > 0 Then msg = msg & vbCrLf & "※ Aにしかない列(比べていません): " & onlyA
    If Len(onlyB) > 0 Then msg = msg & vbCrLf & "※ Bにしかない列(比べていません): " & onlyB
    If tol > 0 Then msg = msg & vbCrLf & "※ 数字の差が " & tol & " 以内は、同じとしました。"
    Set Recon_Do = ws
    Exit Function
EH:
    Application.ScreenUpdating = True
    msg = "うまくいきませんでした。" & vbCrLf & Err.Description
    If Not ws Is Nothing Then
        Application.DisplayAlerts = False
        ws.Delete
        Application.DisplayAlerts = True
    End If
End Function

Public Function BT_MoveToA1(ByVal wb As Workbook) As Long
    Dim ws As Object, firstWs As Object, skipped As Long
    On Error Resume Next
    wb.Activate
    On Error GoTo 0
    For Each ws In wb.Sheets
        If ws.Visible = xlSheetVisible Then
            If firstWs Is Nothing Then Set firstWs = ws
            On Error Resume Next
            Err.Clear
            ws.Activate
            If TypeName(ws) = "Worksheet" Then
                ActiveWindow.ScrollRow = 1
                ActiveWindow.ScrollColumn = 1
                ws.Range("A1").Select
            End If
            If Err.Number <> 0 Then skipped = skipped + 1
            On Error GoTo 0
        End If
    Next ws
    If Not firstWs Is Nothing Then
        On Error Resume Next
        firstWs.Activate
        On Error GoTo 0
    End If
    BT_MoveToA1 = skipped
End Function

' ---- 2. 空白行の削除 ----
' データの中にある、すべてのセルが空の行を、下から削除する。削除した行数を返す
' (数式が入っている行は、空白とみなさない。データより上の空行は、触らない)
Public Function BT_DeleteBlankRows(ByVal ws As Worksheet) As Long
    Dim ur As Range, r As Long, firstRow As Long, lastRow As Long, c1 As Long, c2 As Long, cnt As Long
    If Application.WorksheetFunction.CountA(ws.UsedRange) = 0 Then Exit Function
    Set ur = ws.UsedRange
    firstRow = ur.Row
    lastRow = ur.Row + ur.Rows.Count - 1
    c1 = ur.Column
    c2 = ur.Column + ur.Columns.Count - 1
    For r = lastRow To firstRow Step -1
        If Application.WorksheetFunction.CountA(ws.Range(ws.Cells(r, c1), ws.Cells(r, c2))) = 0 Then
            ws.Rows(r).Delete
            cnt = cnt + 1
        End If
    Next r
    BT_DeleteBlankRows = cnt
End Function
